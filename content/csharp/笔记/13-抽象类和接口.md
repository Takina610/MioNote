# 抽象类

```csharp
using System;

namespace ConsoleApp1
{
    abstract class Animal
    {
        public Animal(string name) {
            Name = name;
        }
        public string Name { get; set; }

        public abstract void speak();
    }

    class Dog: Animal
    {
        public Dog(string name) : base(name)
        {
        }

        public override void speak()
        {
            Console.WriteLine($"{Name}在汪汪叫");
        }
    }

    class Program
    {
        static void Main()
        {
            Animal dog = new Dog("zs");
            dog.speak();
        }
    }
}
```

# 接口

```csharp
using System;

namespace ConsoleApp1
{
    interface IPay
    {
        void pay(int money);
    }

    class AliPay : IPay
    {
        public void pay(int money)
        {
            Console.WriteLine($"使用支付宝支付{money}");
        }
    }

    class WechatPay : IPay
    {
        public void pay(int money)
        {
            Console.WriteLine($"使用微信支付{money}");
        }
    }

    class Program
    {
        static void pay(IPay payWay, int money)
        {
            payWay.pay(money);
        }
        static void Main()
        {
            IPay aliPay = new AliPay();
            IPay wechatPay = new WechatPay();

            pay(aliPay, 100);
            pay(wechatPay, 200);
        }
    }
}
```
