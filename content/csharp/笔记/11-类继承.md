```csharp
using System;

namespace ConsoleApp1
{
    class Animal
    {
        public Animal(string name) {
            Name = name;
        }
        public string Name { get; set; }

        public void eat(string name)
        {
            Console.WriteLine($"{name}在吃");
        }
    }

    class Dog: Animal
    {
        public Dog(string name) : base(name)
        {
        }

        public void speak(string name)
        {
            Console.WriteLine($"{name}在叫");
        }
    }

    class Cat : Animal
    {
        public Cat(string name) : base(name)
        {
        }

        public void speak(string name)
        {
            Console.WriteLine($"{name}在叫");
        }
    }

    sealed class NoSon{}

    class Program
    {
        static void Main()
        {
            Dog dog1 = new Dog("zs");
            Cat cat1 = new Cat("ls");

            Animal dog2 = new Dog("ww");
            Animal cat2 = new Cat("zl");
        }
    }
}
```

- c# 中使用 `:` 继承类
- `: base(name)` 可用于调用基类构造函数
- 被 `sealed` 修饰的类是密封类，被 `sealead` 修饰后，则不能被继承
