```csharp
using System;

namespace ConsoleApp1
{
    class Vector3
    {
        double x, y, z;

        public Vector3 (double x, double y, double z)
        {
            this.x = x;
            this.y = y;
            this.z = z;
        }

        public override string ToString()
        {
            return $"x:{x}, y:{y}, z:{z}";
        }

        public static implicit operator Vector2(Vector3 v) {
            return new(v.x, v.y);
        }
    }

    class Vector2
    {
        double x, y;

        public Vector2(double x, double y)
        {
            this.x = x;
            this.y = y;
        }

        public override string ToString()
        {
            return $"x:{x}, y:{y}";
        }

        public static explicit operator Vector3(Vector2 v) {
            return new(v.x, v.y, 0);
        }
    }

    class Program
    {
        static void Main()
        {
            Vector3 vector3 = new(1, 2, 3);
            Vector2 vector2 = new(4, 5);

            vector3 = (Vector3) vector2;

            vector2 = vector3;
        }
    }
}
```

`implicit` 和 `explicit` 关键字用于定义类型转换运算符。它们允许你自定义你的类（或结构体）如何与其他类型进行互相转换。
- `implicit`（隐式转换）：不需要在代码中强制转换，转换过程自动发生。要求绝不丢失数据且不会抛出异常（安全、无损耗）
- `explicit`（显式转换）：需要在代码中强制转换（使用 (类型) 语法）。可能丢失精度或抛出异常（有风险、有损耗）。

```csharp
public class 源类型
{
    // 隐式转换：从 源类型 -> 目标类型
    public static implicit operator 目标类型(源类型 instance)
    {
        // 返回目标类型的实例
    }

    // 显式转换：从 源类型 -> 目标类型
    public static explicit operator 目标类型(源类型 instance)
    {
        // 返回目标类型的实例，可能抛异常
    }
}
```
